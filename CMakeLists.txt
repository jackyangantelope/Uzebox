     NOT UZEBOX_NATIVE_TRACE_WAYS MATCHES "^[1-9][0-9]*$")
    message(FATAL_ERROR "Native trace cache slots and ways must be positive integers")
  endif()
  math(EXPR UZEBOX_NATIVE_TRACE_REMAINDER "${UZEBOX_NATIVE_TRACE_SLOTS} % ${UZEBOX_NATIVE_TRACE_WAYS}")
  math(EXPR UZEBOX_NATIVE_TRACE_SETS "${UZEBOX_NATIVE_TRACE_SLOTS} / ${UZEBOX_NATIVE_TRACE_WAYS}")
  math(EXPR UZEBOX_NATIVE_TRACE_SET_MASK "${UZEBOX_NATIVE_TRACE_SETS} & (${UZEBOX_NATIVE_TRACE_SETS} - 1)")
  if(UZEBOX_NATIVE_TRACE_SLOTS GREATER 255 OR UZEBOX_NATIVE_TRACE_REMAINDER OR
     UZEBOX_NATIVE_TRACE_SET_MASK)
    message(FATAL_ERROR "Native trace cache requires <=255 slots and a power-of-two set count")
  endif()
  target_compile_definitions(cuzebox_runtime PRIVATE CU_AVR_NATIVE_TRACE=1)
  target_compile_definitions(cuzebox_runtime PRIVATE
    CU_AVR_NTRACE_SLOTS=${UZEBOX_NATIVE_TRACE_SLOTS}
    CU_AVR_NTRACE_WAYS=${UZEBOX_NATIVE_TRACE_WAYS})
